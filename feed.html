-- ISSUES TABLE CREATE TABLE IF NOT EXISTS issues ( id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY, category TEXT NOT NULL, location TEXT NOT NULL, description TEXT NOT NULL, media_url TEXT, upvotes INT DEFAULT 0, status TEXT DEFAULT 'Reported', created_at TIMESTAMP WITH TIME ZONE DEFAULT now(), last_activity TIMESTAMP WITH TIME ZONE DEFAULT now() ); -- COMMENTS CREATE TABLE IF NOT EXISTS comments ( id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY, issue_id BIGINT REFERENCES issues(id) ON DELETE CASCADE, text TEXT NOT NULL, created_at TIMESTAMP WITH TIME ZONE DEFAULT now() ); -- AUTO UPDATE ACTIVITY CREATE OR REPLACE FUNCTION update_last_activity() RETURNS TRIGGER AS $$ BEGIN NEW.last_activity = now(); RETURN NEW; END; $$ LANGUAGE plpgsql; DROP TRIGGER IF EXISTS trg_update_last_activity ON issues; CREATE TRIGGER trg_update_last_activity BEFORE UPDATE ON issues FOR EACH ROW EXECUTE FUNCTION update_last_activity;
